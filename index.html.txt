    <script>
        let currentSlide = 1;
        const totalSlides = 6;

        function updateSlides() {
            for (let i = 1; i <= totalSlides; i++) {
                const slide = document.getElementById(`slide${i}`);
                if (i === currentSlide) {
                    slide.classList.add('active');
                } else {
                    slide.classList.remove('active');
                }
            }

            document.getElementById('prevBtn').disabled = currentSlide === 1;
            
            const nextBtn = document.getElementById('nextBtn');
            if (currentSlide === totalSlides) {
                nextBtn.innerHTML = 'Celebrate Again 🎉';
            } else {
                nextBtn.innerHTML = 'Next <i class="fa-solid fa-chevron-right"></i>';
            }

            const dots = document.getElementById('dotContainer').children;
            for (let i = 0; i < dots.length; i++) {
                if (i === currentSlide - 1) {
                    dots[i].className = 'w-6 h-3 rounded-full bg-purple-600 transition-all';
                } else {
                    dots[i].className = 'w-3 h-3 rounded-full bg-gray-300 transition-all';
                }
            }
        }

        function changeSlide(direction) {
            if (currentSlide === totalSlides && direction === 1) {
                currentSlide = 1;
                triggerConfetti();
            } else {
                currentSlide += direction;
                if (currentSlide < 1) currentSlide = 1;
                if (currentSlide > totalSlides) currentSlide = totalSlides;
            }
            updateSlides();
        }

        /* Escaping No Button Handler */
        function runawayNo() {
            const btn = document.getElementById('noBtn');
            const container = document.getElementById('buttonArea');
            
            const containerRect = container.getBoundingClientRect();
            const btnRect = btn.getBoundingClientRect();

            const maxX = containerRect.width - btnRect.width;

            const randomX = Math.floor(Math.random() * Math.max(maxX, 120)) - (maxX / 2);
            const randomY = Math.floor(Math.random() * 80) - 40;

            btn.style.transform = `translate(${randomX}px, ${randomY}px)`;
        }

        function forgiveAccepted() {
            triggerConfetti();
            document.getElementById('forgiveModal').classList.remove('hidden');
        }

        function closeModal() {
            document.getElementById('forgiveModal').classList.add('hidden');
            changeSlide(1);
        }

        /* Floating Balloons */
        function createBalloons() {
            const container = document.getElementById('balloonContainer');
            const colors = ['#ff7eb3', '#ff758c', '#4299e1', '#9f7aea', '#f6ad55', '#48bb78'];
            
            for (let i = 0; i < 15; i++) {
                const balloon = document.createElement('div');
                balloon.className = 'balloon';
                
                const size = Math.floor(Math.random() * 30) + 40;
                const color = colors[Math.floor(Math.random() * colors.length)];
                const left = Math.random() * 100;
                const duration = Math.random() * 6 + 8;
                const delay = Math.random() * 5;

                balloon.innerHTML = `
                    <svg width="${size}" height="${size * 1.2}" viewBox="0 0 50 60">
                        <ellipse cx="25" cy="25" rx="20" ry="24" fill="${color}" />
                        <polygon points="25,49 20,55 30,55" fill="${color}" />
                        <line x1="25" y1="55" x2="25" y2="70" stroke="#888" stroke-width="1.5"/>
                    </svg>
                `;

                balloon.style.left = `${left}%`;
                balloon.style.animationDuration = `${duration}s`;
                balloon.style.animationDelay = `${delay}s`;

                container.appendChild(balloon);
            }
        }

        /* Web Audio API Synthesizer */
        let audioCtx = null;
        let isPlayingMusic = false;
        let musicInterval = null;

        const melody = [
            { note: 264, duration: 350 }, { note: 264, duration: 350 },
            { note: 297, duration: 700 }, { note: 264, duration: 700 },
            { note: 352, duration: 700 }, { note: 330, duration: 1200 },
            { note: 264, duration: 350 }, { note: 264, duration: 350 },
            { note: 297, duration: 700 }, { note: 264, duration: 700 },
            { note: 396, duration: 700 }, { note: 352, duration: 1200 }
        ];

        function playNote(freq, duration) {
            if (!audioCtx) return;
            try {
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
                gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + (duration / 1000));
                
                osc.connect(gain);
                gain.connect(audioCtx.destination);
                
                osc.start();
                osc.stop(audioCtx.currentTime + (duration / 1000));
            } catch (e) {
                console.log(e);
            }
        }

        function startMelody() {
            let noteIdx = 0;
            isPlayingMusic = true;
            document.getElementById('musicText').innerText = 'Pause Tune ⏸️';
            
            function playNext() {
                if (!isPlayingMusic) return;
                const current = melody[noteIdx];
                playNote(current.note, current.duration);
                noteIdx = (noteIdx + 1) % melody.length;
                musicInterval = setTimeout(playNext, current.duration + 50);
            }
            playNext();
        }

        function stopMelody() {
            isPlayingMusic = false;
            clearTimeout(musicInterval);
            document.getElementById('musicText').innerText = 'Play B\'day Tune 🎵';
        }

        function toggleMusic() {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }
            if (audioCtx.state === 'suspended') {
                audioCtx.resume();
            }

            if (isPlayingMusic) {
                stopMelody();
            } else {
                startMelody();
            }
        }

        window.onload = function() {
            updateSlides();
            createBalloons();
            triggerConfetti();

            document.body.addEventListener('click', function initAudio() {
                if (!audioCtx) {
                    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                }
                document.body.removeEventListener('click', initAudio);
            }, { once: true });
        };
